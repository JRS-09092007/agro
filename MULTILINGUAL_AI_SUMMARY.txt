💻 CODE EXAMPLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Example 1: Using Translations in UI
────────────────────────────────────

final appProvider = Provider.of<AppProvider>(context);

Text(appProvider.t('hello'))  // नमस्ते
Text(appProvider.t('dashboard'))  // डैशबोर्ड
Text(appProvider.t('crop_recommendation'))  // फसल की सिफारिश


Example 2: AI Response in Farmer's Language
────────────────────────────────────────────

final response = apiService.generateAIResponse(
  appProvider.language,  // 'hi' for Hindi, 'ta' for Tamil, etc.
  'crop_recommendation',
  {
    'crop': 'Rice',
    'soil': 'Loamy',
    'season': 'Monsoon',
    'water': '120-150 cm'
  }
);

// Output in Hindi:
// "आपकी मिट्टी के प्रकार (Loamy) और Monsoon ऋतु के आधार पर,
//  मैं Rice उगाने की सिफारिश करता हूं। इसके लिए लगभग 120-150 cm
//  पानी की आवश्यकता है।"


Example 3: Language Switching
──────────────────────────────

// User taps language selector
onLanguageChange(String newLanguage) {
  appProvider.setLanguage(newLanguage);
  // Entire app rebuilds with new language
  // Preference saved to device
}


Example 4: Translate Any Language
──────────────────────────────────

// Get translation in specific language (without changing current)
final hindiText = appProvider.tInLanguage('crop_recommendation', 'hi');
final tamilText = appProvider.tInLanguage('crop_recommendation', 'ta');
final marathiText = appProvider.tInLanguage('crop_recommendation', 'mr');
